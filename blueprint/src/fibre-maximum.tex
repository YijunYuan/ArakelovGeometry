\subsection{Maxima over finite extension fibres}

Throughout this subsection $S=(K,(\Omega,\mathcal A,\nu),\phi)$ is an
adelic curve and $L/K$ is a finite extension, with no separability or
normality hypothesis. Write $X_\omega=\pi_{L/K}^{-1}(\{\omega\})$
inside the intrinsic parameter space $\Omega\times_{M_K}M_L$.
For the finite integration operator, write $E$ for the separable part of
$L/K$, $\Omega_L^{\mathrm{fin}}=\Omega_E$ for its parameter model, and
$X_\omega^{\mathrm{fin}}$ for its fibre. Unique extension over $L/E$
gives the base-preserving measurable identification
$e:\Omega_L^{\mathrm{fin}}\to\Omega_L$.

\begin{definition}[Fibre maximum]
\label{def:fibre-maximum}
\leanok
\source{chen-moriwaki-adelic:page-0218}
\uses{lem:finite-extension-fibres,prop:algebraic-restriction-surjective}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.nonempty_extensionFiber,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.lt_fibreMaximum_iff,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreMaximum,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.exists_eq_fibreMaximum,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.le_fibreMaximum,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreMaximum_le_iff}
For a function $F:\Omega_L\to\mathbb R$, define
\[
 \mathcal M_{L/K}F(\omega)=\max_{x\in X_\omega}F(x).
\]
Each fibre is finite and nonempty, so this is a finite real number and
the maximum is attained. It is the least upper bound of the values of
$F$ on $X_\omega$.
\end{definition}
\begin{proof}
Extension of the absolute value $\phi_\omega$ to $L$ gives a point of
$X_\omega$, and finiteness of extensions makes $X_\omega$ finite.
A real function on a finite nonempty set has an attaining maximum:
start with one value and successively replace it by the larger of it
and each of the remaining finitely many values. The resulting value
belongs to the range, dominates every value, and is below any upper
bound of that range.
\end{proof}

\begin{lemma}[Positive fibre mass detects nonempty intersection]
\label{lem:finite-fibre-indicator-positive}
\leanok
\source{chen-moriwaki-adelic:page-0218,chen-moriwaki-adelic:page-0219}
\uses{lem:finite-fibre-sum,def:finite-fibre-lintegral}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceFinExtension.fibreLIntegral_indicator_pos_iff}
For any subset $B\subseteq\Omega_L^{\mathrm{fin}}$ and any
$\omega\in\Omega$,
\[
 I^{\mathrm{fin},+}_{L/K}(\mathbf1_B)(\omega)>0
 \quad\Longleftrightarrow\quad B\cap X_\omega^{\mathrm{fin}}\ne\varnothing.
\]
\end{lemma}
\begin{proof}
Fibre integration is the finite sum of the values of $\mathbf1_B$
multiplied by the point masses. Each point mass is strictly positive.
Every term vanishes when $B$ misses the fibre; a point in their
intersection contributes a strictly positive term, and every other
term is nonnegative. This proves both implications.
\end{proof}

\begin{lemma}[Measurable images under a finite extension projection]
\label{lem:finite-projection-measurable-image}
\leanok
\source{chen-moriwaki-adelic:page-0218,chen-moriwaki-adelic:page-0219}
\uses{lem:finite-parameter-identification,lem:finite-fibre-indicator-positive,lem:finite-kernel-measurable}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.measurableSet_image_proj}
If $A\subseteq\Omega_L$ is measurable, then $\pi_{L/K}(A)$ is
$\mathcal A$-measurable.
\end{lemma}
\begin{proof}
Let $E$ be the separable part of $L/K$, and let
$e:\Omega_E\to\Omega_L$ be the measurable bijection extending absolute
values uniquely over $L/E$. It preserves the base point and hence
identifies the fibres over each $\omega$. Set $B=e^{-1}(A)$.
The function
\[
 g(\omega)=\int_{X_{E,\omega}}\mathbf1_B(x)\,dP_{E,\omega}(x)
\]
is measurable by the measurability of finite fibre integration.
Every point of $X_{E,\omega}$ has strictly positive probability mass.
Thus the displayed finite sum vanishes if $B$ misses the fibre; if
$x$ belongs to their intersection, its summand is the positive mass
of $x$, so the sum is positive. Consequently
$\pi_{L/K}(A)=\{\omega:g(\omega)>0\}$, a measurable set.
\end{proof}

\begin{proposition}[Measurability of the fibre maximum]
\label{prop:fibre-maximum-measurable}
\leanok
\source{chen-moriwaki-adelic:page-0218,chen-moriwaki-adelic:page-0219}
\dcref{Proposition 3.6.2}
\uses{def:fibre-maximum,lem:finite-projection-measurable-image,prop:projective-measurable-equality}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.measurable_fibreMaximum,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.measurable_fibreMaximum_projective}
For every $\mathcal A_L$-measurable function $F:\Omega_L\to\mathbb R$,
the function $\mathcal M_{L/K}F:\Omega\to\mathbb R$ is measurable.
Here $\mathcal A_L$ is the sigma algebra of $S\otimes_K L$, equivalently
the sigma algebra generated by the base projection and all evaluations.
\end{proposition}
\begin{proof}
For every real $c$, attainment of the maximum gives
\[
 \{\omega:c<\mathcal M_{L/K}F(\omega)\}
 =\pi_{L/K}(\{x:c<F(x)\}).
\]
The right side is measurable by the preceding lemma. Strict upper
rays generate the Borel sigma algebra of $\mathbb R$, so these
measurable inverse images prove the assertion. Proposition 3.4.13
identifies the evaluation sigma algebra with the projective sigma
algebra of $S\otimes_K L$.
\end{proof}

\begin{remark}
Proposition 3.6.2 of \cite{chen-moriwaki} first treats a normal extension
by automorphism orbits and then passes to a finite normal overfield.
The argument above uses instead the already constructed finite fibre
probabilities, whose every point has positive mass, and their
measurable integration operator. The canonical identification with the
separable part carries this argument to every finite extension and
retains exactly the proposition's hypotheses and conclusion.
\end{remark}
