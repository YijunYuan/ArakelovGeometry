\section{Parameter spaces and fibre integration}

Let $K$ be a field with adelic structure $S=(\Omega,\mathcal A,\nu,\phi)$.
Thus every $\phi_\omega$ is a real absolute value on $K$, and for $a\ne0$
the function $\omega\mapsto\log\phi_\omega(a)$ is measurable and integrable.
The construction follows \cite[\S\S3.3--3.4]{chen-moriwaki}.

\begin{definition}[Extension parameter space]\label{def:extension-parameters}
\source{chen-moriwaki-adelic:page-0194}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.instMeasurableSpace}
For a field extension $L/K$ put
\[
 \Omega_L=\{(\omega,w):\omega\in\Omega,\quad
 w\text{ an absolute value on }L,\quad w|_K=\phi_\omega\}.
\]
Write $\pi(\omega,w)=\omega$ and $e_a(\omega,w)=w(a)$ for $a\in L$.
Equip $\Omega_L$ with the smallest sigma algebra making $\pi$ and all $e_a$
measurable. The fibre $X_\omega=\pi^{-1}\{\omega\}$ carries the induced
sigma algebra. A singleton of $\Omega$ need not be measurable; the induced
sigma algebra on $X_\omega$ is nevertheless defined.
\end{definition}

\begin{lemma}[Measurable restriction]\label{lem:measurable-restriction}
\uses{def:extension-parameters}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.measurable_restrict,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.measurable_restrictOfLE}
If $F\subseteq G\subseteq L$ are intermediate fields, restriction
$\Omega_L\to\Omega_F$ and restriction $\Omega_G\to\Omega_F$ are measurable.
\end{lemma}
\begin{proof}
Restriction preserves the base parameter. Composing an evaluation at $a\in F$
with restriction is evaluation at the image of $a$ in the larger field.
The preimages of both kinds of generating measurable sets are therefore
measurable, which proves the assertion for their generated sigma algebra.
\end{proof}

\begin{lemma}[Transitivity of restriction]\label{lem:restriction-transitive}
\uses{def:extension-parameters}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.restrictOfLE_restrict}
For $F\subseteq G\subseteq L$, restriction from $L$ to $G$ and then to $F$
equals restriction from $L$ to $F$.
\end{lemma}
\begin{proof}
Both maps preserve $\omega$. Their absolute values at $a\in F$ are both
$w(a)$, because the composite of the two field inclusions is the direct
inclusion. Equality of the base parameters and of all absolute-value
evaluations gives equality of the resulting pairs.
\end{proof}

\begin{lemma}[Fibre separation]\label{lem:fibre-separation}
\uses{def:extension-parameters}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.extensionFiber_separatesPoints}
The induced measurable space on every $X_\omega$ separates points.
\end{lemma}
\begin{proof}
Distinct points in $X_\omega$ have the same base parameter and therefore
different absolute values $w,w'$. Choose $a\in L$ with $w(a)\ne w'(a)$.
The inverse image of the closed singleton $\{w(a)\}$ under $e_a|_{X_\omega}$
is measurable, contains the first point and does not contain the second.
\end{proof}

\begin{definition}[Weighted fibre integral]\label{def:weighted-fibre-integral}
\source{chen-moriwaki-adelic:page-0194}
\uses{def:extension-parameters}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreWeight,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreProbabilityMeasure,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreIntegral,ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreLIntegral}
Suppose $L/K$ is finite separable. For $x=(\omega,w)\in X_\omega$ let
$L_w$ and $K_{w|_K}$ denote the completions and set
\[
 p_\omega(x)=\frac{[L_w:K_{w|_K}]}{[L:K]},\qquad
 P_\omega=\sum_{x\in X_\omega}p_\omega(x)\delta_x.
\]
The nonnegative fibre integral is
$I^+f(\omega)=\int_{X_\omega} f\,dP_\omega\in[0,+\infty]$.
For real-valued $f$, write $If(\omega)=\int_{X_\omega}f\,dP_\omega$,
with the convention that a nonintegrable Bochner integral is zero.
The following two identities do not require the normalization of $P_\omega$.
\end{definition}

\begin{lemma}[Homogeneity]\label{lem:fibre-homogeneity}
\source{chen-moriwaki-adelic:page-0195}
\uses{def:weighted-fibre-integral}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreIntegral_smul}
For $c\in\mathbb R$ and any real-valued function $f$ on $\Omega_L$,
$I(cf)=c\,If$.
\end{lemma}
\begin{proof}
Fix $\omega$. If $c=0$, both sides vanish. If $c\ne0$, multiplication by
$c$ preserves and reflects integrability. If $f|_{X_\omega}$ is integrable,
linearity of the Bochner integral gives the equality. Otherwise both integrals
in the asserted equality are zero by the stated convention.
\end{proof}

\begin{proposition}[Multiplication by a base function]\label{prop:fibre-base-multiplication}
\source{chen-moriwaki-adelic:page-0195}
\uses{def:weighted-fibre-integral,lem:fibre-homogeneity}
\lean{ArakelovGeometry.AdelicStructure.ParameterSpaceExtension.fibreIntegral_comp_proj_mul}
For functions $\varphi:\Omega\to\mathbb R$ and $f:\Omega_L\to\mathbb R$,
\[
 I((\varphi\circ\pi)f)(\omega)=\varphi(\omega)If(\omega).
\]
\end{proposition}
\begin{proof}
For $x\in X_\omega$ one has $\pi(x)=\omega$, so the integrand restricted
to the fibre is $\varphi(\omega)f(x)$. Apply homogeneity on this fibre with
the constant scalar $\varphi(\omega)$.
\end{proof}
