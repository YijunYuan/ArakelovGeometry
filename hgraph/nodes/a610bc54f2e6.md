private theorem uniformBound_mul_right {u : ℕ → X → ℝ} {g : X → ℝ}
    (hu : ∃ M : ℝ, ∀ n x, |u n x| ≤ M) (hg : ∃ N : ℝ, ∀ x, |g x| ≤ N) :
