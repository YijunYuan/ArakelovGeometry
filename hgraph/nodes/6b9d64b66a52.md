theorem eq_ofSingletons {f : X → ℝ≥0∞} {μ : Measure X} (hμ : ∀ x, μ {x} = f x) :
    μ = ofSingletons f :=
  Measure.ext_of_singleton fun x ↦ by rw [hμ, ofSingletons_singleton]
