noncomputable def projectiveHeight (x : Projectivization K (ι → K)) : ℝ :=
  x.lift (fun a ↦ AS.logHeight a.val) fun a b t h ↦ by
    have ht : t ≠ 0 := by
      intro ht
      exact a.property (by simpa [ht] using h)
    rw [h, AS.logHeight_smul_eq b.property ht]
