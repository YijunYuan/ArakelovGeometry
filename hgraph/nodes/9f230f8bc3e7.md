instance instInfinitePlace {K : Type*} [Field K] [NumberField K] : Infinite (Place K) := by
  have hFinite : Infinite (FinitePlace K) :=
    (Equiv.infinite_iff FinitePlace.equivHeightOneSpectrum).mpr inferInstance
