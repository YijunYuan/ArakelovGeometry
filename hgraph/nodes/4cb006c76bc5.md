theorem pointDegree_eq_of_span {q : Polynomial k}
    {v : HeightOneSpectrum (Polynomial k)} (hv : v.asIdeal = Ideal.span {q}) :
    pointDegree v = q.natDegree := by
  obtain ⟨u, hu⟩ : Associated (Submodule.IsPrincipal.generator v.asIdeal) q := by
    rw [← Ideal.span_singleton_eq_span_singleton, Ideal.span_singleton_generator, hv]
  have hgen : Submodule.IsPrincipal.generator v.asIdeal ≠ 0 := by
    intro h0
    exact v.ne_bot (by rw [← Ideal.span_singleton_generator v.asIdeal, h0,
      Ideal.span_singleton_eq_bot])
  rw [pointDegree, ← hu, Polynomial.natDegree_mul hgen u.ne_zero,
    Polynomial.natDegree_eq_zero_of_isUnit u.isUnit, add_zero]
