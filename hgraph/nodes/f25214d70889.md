theorem mk_normalizedCoordinates (x : Projectivization L (ι → L))
    {j : ι} (hj : x.rep j ≠ 0) :
    mk L (x.normalizedCoordinates j) (x.normalizedCoordinates_ne_zero hj) = x := by
  calc
    _ = mk L x.rep x.rep_nonzero := by
      apply (mk_eq_mk_iff' L _ _ _ _).mpr
      refine ⟨(x.rep j)⁻¹, ?_⟩
      ext i
      simp [normalizedCoordinates, div_eq_mul_inv, mul_comm]
    _ = x := x.mk_rep
