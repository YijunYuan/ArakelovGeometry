theorem finsum_weighted_log_prime {q : Polynomial k} (hq : Prime q) :
    (∑ᶠ ω, (pointWeight k ω).toReal *
      Real.log (pointAbv k ω (algebraMap (Polynomial k) (RatFunc k) q))) = 0 := by
  classical
  have hq0 : algebraMap (Polynomial k) (RatFunc k) q ≠ 0 :=
    RatFunc.algebraMap_ne_zero hq.ne_zero
  rw [finsum_eq_sum_of_support_subset _ (s := {some (spanPrime hq), none}) ?_]
  · rw [Finset.sum_pair (by simp)]
    rw [pointWeight_some, pointAbv_some, pointWeight_none, pointAbv_none,
      real_log_adicAbv _ hq0, valuation_spanPrime_self hq, log_exp,
      real_log_inftyAbv_algebraMap hq.ne_zero, pointDegree_spanPrime hq,
      ENNReal.toReal_natCast, ENNReal.toReal_one, one_mul]
    push_cast
    ring
  · rintro (_ | v) hω
    · simp
    · simp only [Finset.coe_insert, Finset.coe_singleton, Set.mem_insert_iff,
        Set.mem_singleton_iff]
      left
      by_contra hv
      have hvne : v ≠ spanPrime hq := fun h ↦ hv (by rw [h])
      simp only [Function.mem_support] at hω
      exact hω (by rw [pointAbv_some, real_log_adicAbv v hq0,
        valuation_algebraMap_eq_one_of_ne hq hvne, log_one, Int.cast_zero, mul_zero])
