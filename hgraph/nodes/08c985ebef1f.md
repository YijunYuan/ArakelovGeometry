theorem finite_support_log {f : RatFunc k} (hf : f ≠ 0) :
    (Function.support fun ω ↦ Real.log (pointAbv k ω f)).Finite := by
  have hnum : (Ideal.span {f.num} : Ideal (Polynomial k)) ≠ 0 := by
    rw [Ne, Ideal.zero_eq_bot, Ideal.span_singleton_eq_bot]
    exact RatFunc.num_ne_zero hf
  have hden : (Ideal.span {f.denom} : Ideal (Polynomial k)) ≠ 0 := by
    rw [Ne, Ideal.zero_eq_bot, Ideal.span_singleton_eq_bot]
    exact f.denom_ne_zero
  exact Set.Finite.subset (Set.Finite.insert none (Set.Finite.image some
    ((Ideal.finite_factors hnum).union (Ideal.finite_factors hden))))
    (support_log_pointAbv_subset f)
