noncomputable def placeWeight (ω : AbsoluteValue.Place K) : ℝ≥0∞ :=
  if ω = trivialPlace K then 0
  else if hna : IsNonarchimedeanPlace ω then 1
  else (((isArchimedeanPlace_iff ω).mp hna).choose.mult : ℝ≥0∞)
