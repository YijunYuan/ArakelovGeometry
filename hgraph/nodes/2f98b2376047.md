    placeRep (Quotient.mk' v.val : AbsoluteValue.Place K) = v.val := by
  unfold placeRep

  · exact absurd h1 (mk_finitePlace_ne_trivialPlace v)

  · exact absurd (isNonarchimedeanPlace_mk_finitePlace v) h2
