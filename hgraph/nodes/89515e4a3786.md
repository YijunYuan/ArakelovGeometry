theorem orbitMaximum_le (F : ParameterSpaceExtension AS L → ℝ)
    (x : ParameterSpaceExtension AS L) {b : ℝ}
    (hb : ∀ τ : L ≃ₐ[K] L, F (aut AS L τ x) ≤ b) : orbitMaximum AS L F x ≤ b :=
  Finset.sup'_le _ _ (fun τ _ ↦ hb τ)
