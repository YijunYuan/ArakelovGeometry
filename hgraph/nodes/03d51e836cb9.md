private theorem κ_pos_of_Ωinf {K : Type*} [Field K] (AS : AdelicStructure K) {ω : AS.Ω}
    (hω : ω ∈ AS.Ωinf) : 0 < AS.κ ω := by
  unfold κ
  rw [dif_pos hω]
  exact (exists_rpow_eq_of_not_isNonarchimedean (AS.ϕ ω) hω.out).choose_spec.1.out.1
