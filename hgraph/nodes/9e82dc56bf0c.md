theorem finrank_adjoin_div_le (x : Projectivization L (ι → L))
    [FiniteDimensional K (fieldOfDefinition K x)] (i j : ι) :
    Module.finrank K (IntermediateField.adjoin K {x.rep i / x.rep j}) ≤
      Module.finrank K (fieldOfDefinition K x) := by
  apply IntermediateField.finrank_le_of_le_right
  exact IntermediateField.adjoin_simple_le_iff.mpr (div_mem_fieldOfDefinition K x i j)

end Projectivization
