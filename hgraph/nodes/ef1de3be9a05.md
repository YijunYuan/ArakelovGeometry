def ofElement (a : L) : FiniteIntermediateField K L :=
  ⟨IntermediateField.adjoin K ({a} : Set L), by
    exact IntermediateField.adjoin.finiteDimensional
      ((Algebra.IsAlgebraic.isAlgebraic (R := K) a).isIntegral)⟩
