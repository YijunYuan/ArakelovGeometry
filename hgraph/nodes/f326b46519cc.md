noncomputable def ofSingletons (f : X → ℝ≥0∞) : Measure X :=
  Measure.sum (fun a ↦ f a • Measure.dirac a)

variable [MeasurableSingletonClass X]

/-- `ofSingletons f` takes the prescribed value `f x` on the singleton `{x}`. -/
