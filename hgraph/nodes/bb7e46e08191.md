theorem valuation_eq_one_of_intValuation_num_denom {f : RatFunc k}
    {v : HeightOneSpectrum (Polynomial k)}
    (hnum : v.intValuation f.num = 1) (hden : v.intValuation f.denom = 1) :
    v.valuation (RatFunc k) f = 1 := by
  have key := congrArg (v.valuation (RatFunc k)) (algebraMap_num_eq_mul_denom f)
  rw [map_mul, valuation_of_algebraMap, valuation_of_algebraMap, hnum, hden, mul_one] at key
  exact key.symm
