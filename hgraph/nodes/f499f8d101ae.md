private def boundedSeqClosed (S : Set (X → ℝ)) : Prop :=
  ∀ ⦃u : ℕ → X → ℝ⦄ ⦃f : X → ℝ⦄, (∀ n, u n ∈ S) →
    (∃ M : ℝ, ∀ n x, |u n x| ≤ M) → Tendsto u atTop (nhds f) → f ∈ S
