theorem fieldOfDefinition_le_iff (x : Projectivization L (ι → L))
    (F : IntermediateField K L) : fieldOfDefinition K x ≤ F ↔
      ∃ (a : ι → L) (ha : a ≠ 0), mk L a ha = x ∧ ∀ i, a i ∈ F := by
  constructor
  · intro h
    obtain ⟨j, hj⟩ : ∃ j, x.rep j ≠ 0 := by
      by_contra! hzero
      exact x.rep_nonzero (funext hzero)
    refine ⟨x.normalizedCoordinates j, x.normalizedCoordinates_ne_zero hj,
      x.mk_normalizedCoordinates hj, ?_⟩
    exact fun i ↦ h (div_mem_fieldOfDefinition K x i j)
  · rintro ⟨a, ha, hax, hF⟩
    rw [← hax, fieldOfDefinition_mk]
    apply IntermediateField.adjoin_le_iff.mpr
    rintro _ ⟨⟨i, j⟩, rfl⟩
    exact F.div_mem (hF i) (hF j)
