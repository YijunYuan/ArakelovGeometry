def primeValuation (p : Nat.Primes) : Valuation (RatFunc ℚ) ℝ≥0 :=
  (primePolynomialValuation p).extendToLocalization
    (S := nonZeroDivisors ℚ[X])
    (fun _ h ↦ primePolynomialValuation_ne_zero p (nonZeroDivisors.ne_zero h))
    (RatFunc ℚ)
