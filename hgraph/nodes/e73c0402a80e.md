  change MeasurableSpace.comap _ (_ ⊔ ⨆ a : separableClosure K F.1, _) ≤ _
  rw [MeasurableSpace.comap_sup, MeasurableSpace.comap_iSup]

    exact (ParameterSpaceExtension.measurable_proj AS (separableClosure K G.1)).comap_le
