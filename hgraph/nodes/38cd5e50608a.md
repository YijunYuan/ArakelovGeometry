theorem v_Ωinf_lt_top_of_κ_bddBelow {K : Type*} [Field K] (AS : AdelicStructure K)
    (hκ : ∃ ε : ℝ, ε > 0 ∧ ε ∈ lowerBounds (AS.κ '' AS.Ωinf)) : AS.v AS.Ωinf < ⊤ := by
  obtain ⟨ε, hε_pos, hlb⟩ := hκ
  -- `κ` is bounded below by `ε > 0` on `Ωinf`, so `Ωinf ⊆ {ω | ε ≤ κ ω}`.
  have hsub : AS.Ωinf ⊆ {ω | ε ≤ AS.κ ω} := fun ω hω ↦ hlb (Set.mem_image_of_mem AS.κ hω)
  -- The latter set has finite measure since `κ` is integrable and `ε > 0`.
  exact lt_of_le_of_lt (measure_mono hsub) <| (AS.integrable_κ).measure_ge_lt_top hε_pos
