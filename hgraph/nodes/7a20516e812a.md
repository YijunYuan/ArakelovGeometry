noncomputable def fieldOfDefinition (x : Projectivization L (ι → L)) :
    IntermediateField K L :=
  x.lift (fun a ↦ coordinateField K a.val) fun a b t h ↦ by
    have ht : t ≠ 0 := by
      intro ht
      exact a.property (by simpa [ht] using h)
    rw [h, coordinateField_smul K _ ht]
