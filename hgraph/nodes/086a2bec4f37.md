theorem finsum_weighted_log_eq_zero {f : RatFunc k} (hf : f ≠ 0) :
    (∑ᶠ ω, (pointWeight k ω).toReal * Real.log (pointAbv k ω f)) = 0 := by
  have hkey : (∑ᶠ ω, (pointWeight k ω).toReal *
      Real.log (pointAbv k ω (algebraMap (Polynomial k) (RatFunc k) f.num))) =
        (∑ᶠ ω, (pointWeight k ω).toReal * Real.log (pointAbv k ω f)) +
          ∑ᶠ ω, (pointWeight k ω).toReal *
            Real.log (pointAbv k ω (algebraMap (Polynomial k) (RatFunc k) f.denom)) := by
    rw [algebraMap_num_eq_mul_denom f]
    exact finsum_weighted_log_mul hf (RatFunc.algebraMap_ne_zero f.denom_ne_zero)
  rw [finsum_weighted_log_polynomial (RatFunc.num_ne_zero hf),
    finsum_weighted_log_polynomial f.denom_ne_zero] at hkey
  linarith

end Points
