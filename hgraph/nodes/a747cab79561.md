noncomputable def coefficientLogBound (p : Polynomial K) : AS.Ω → ℝ :=
  (Finset.range (p.natDegree + 1)).sup'
    (Finset.nonempty_range_iff.mpr (Nat.succ_ne_zero _))
    (fun i ω ↦ Real.log (AS.ϕ ω (p.coeff i)))
