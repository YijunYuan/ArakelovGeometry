theorem existsUnique_singleton (f : X → ℝ≥0∞) : ∃! μ : Measure X, ∀ x, μ {x} = f x :=
  ⟨ofSingletons f, ofSingletons_singleton f, fun _ hν ↦ eq_ofSingletons hν⟩
