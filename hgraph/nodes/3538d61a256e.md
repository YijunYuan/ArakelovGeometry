theorem finsum_weighted_log_mul {f g : RatFunc k} (hf : f ≠ 0) (hg : g ≠ 0) :
    (∑ᶠ ω, (pointWeight k ω).toReal * Real.log (pointAbv k ω (f * g))) =
      (∑ᶠ ω, (pointWeight k ω).toReal * Real.log (pointAbv k ω f)) +
        ∑ᶠ ω, (pointWeight k ω).toReal * Real.log (pointAbv k ω g) := by
  rw [← finsum_add_distrib (finite_support_weighted_log hf) (finite_support_weighted_log hg)]
  exact finsum_congr fun ω ↦ by
    rw [map_mul, Real.log_mul ((pointAbv k ω).ne_zero hf) ((pointAbv k ω).ne_zero hg),
      mul_add]
