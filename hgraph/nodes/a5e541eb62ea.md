theorem finsum_weighted_log_polynomial :
    ∀ {p : Polynomial k}, p ≠ 0 →
      (∑ᶠ ω, (pointWeight k ω).toReal *
        Real.log (pointAbv k ω (algebraMap (Polynomial k) (RatFunc k) p))) = 0 := by
  intro p
  induction p using UniqueFactorizationMonoid.induction_on_prime with
  | h₁ => exact fun h ↦ absurd rfl h
  | h₂ x hx =>
    intro _
    obtain ⟨c, hcu, rfl⟩ := Polynomial.isUnit_iff.mp hx
    exact finsum_weighted_log_C hcu.ne_zero
  | h₃ a q ha hq ih =>
    intro _
    rw [map_mul, finsum_weighted_log_mul (RatFunc.algebraMap_ne_zero hq.ne_zero)
      (RatFunc.algebraMap_ne_zero ha), finsum_weighted_log_prime hq, ih ha, add_zero]
