theorem support_log_pointAbv_subset (f : RatFunc k) :
    (Function.support fun ω ↦ Real.log (pointAbv k ω f)) ⊆
      insert none (some '' ({v : HeightOneSpectrum (Polynomial k) |
        v.asIdeal ∣ Ideal.span {f.num}} ∪
          {v : HeightOneSpectrum (Polynomial k) | v.asIdeal ∣ Ideal.span {f.denom}})) := by
  rintro (_ | v) hω
  · exact Set.mem_insert _ _
  · refine Set.mem_insert_of_mem _ ⟨v, ?_, rfl⟩
    by_contra hv
    rw [Set.mem_union, not_or] at hv
    simp only [Function.mem_support] at hω
    by_cases hf : f = 0
    · exact hω (by simp [hf])
    · exact hω (by rw [pointAbv_some, real_log_adicAbv v hf,
        valuation_eq_one_of_intValuation_num_denom
          (intValuation_eq_one_of_not_dvd hv.1) (intValuation_eq_one_of_not_dvd hv.2),
        log_one, Int.cast_zero])
