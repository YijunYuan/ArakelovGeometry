theorem measurable_coefficientLogBound (p : Polynomial K) :
    Measurable (coefficientLogBound AS p) := by
  apply Finset.measurable_sup'
  intro i _
  by_cases hi : p.coeff i = 0
  · simp [hi]
  · exact AS.measurable_log _ hi
