def fibreCylinders (ω : AS.Ω) : Set (Set (BaseFiber L ω)) :=
  {A | ∃ (F : FiniteIntermediateField K L) (B : Set (FiniteBaseFiber L F ω)),
    projFiniteFiber F ω ⁻¹' B = A}
