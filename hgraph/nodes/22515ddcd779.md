theorem κ_mem_Icc {K : Type*} [Field K] (AS : AdelicStructure K) (ω : AS.Ω) :
    AS.κ ω ∈ Set.Icc 0 1 := by
  by_cases hω : ω ∈ AS.Ωinf
  · unfold κ
    rw [dif_pos hω]
    exact Set.mem_Icc_of_Ioc
      (exists_rpow_eq_of_not_isNonarchimedean (AS.ϕ ω) hω.out).choose_spec.1
  · simp [κ, hω]
