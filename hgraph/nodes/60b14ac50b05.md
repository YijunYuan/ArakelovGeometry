def boundedEvaluationGenerators : Set (ParameterSpaceExtension AS L → ℝ) :=
  {f | (∃ M : ℝ, ∀ x, |f x| ≤ M) ∧
    ∃ b : L, ∃ φ : AS.Ω → ℝ, Measurable φ ∧ (∀ ω, 0 ≤ φ ω) ∧
      (∃ M : ℝ, ∀ ω, |φ ω| ≤ M) ∧ f = fun x ↦ x.ϕ b * φ x.proj}
