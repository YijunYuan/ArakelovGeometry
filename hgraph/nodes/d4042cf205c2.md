theorem measurable_eval (a : K) : Measurable (fun ω : AS.Ω ↦ AS.ϕ ω a) := by
  by_cases ha : a = 0
  · simp [ha]
  · have he : (fun ω : AS.Ω ↦ AS.ϕ ω a) =
        (fun ω : AS.Ω ↦ Real.exp (Real.log (AS.ϕ ω a))) := by
      funext ω
      exact (Real.exp_log ((AS.ϕ ω).pos ha)).symm
    rw [he]
    exact (AS.measurable_log a ha).exp
