    (∑ᶠ ω, (pointWeight k ω).toReal *
      Real.log (pointAbv k ω (algebraMap (Polynomial k) (RatFunc k) (Polynomial.C c)))) = 0 := by
  have hzero : ∀ ω, Real.log (pointAbv k ω
      (algebraMap (Polynomial k) (RatFunc k) (Polynomial.C c))) = 0 := by
    rintro (_ | v)
    · rw [pointAbv_none, real_log_inftyAbv_algebraMap (Polynomial.C_ne_zero.mpr hc),
        Polynomial.natDegree_C, Nat.cast_zero]
    · rw [pointAbv_some, real_log_adicAbv v
        (RatFunc.algebraMap_ne_zero (Polynomial.C_ne_zero.mpr hc)),
        valuation_algebraMap_C v hc, log_one, Int.cast_zero]
  simp only [hzero, mul_zero]
