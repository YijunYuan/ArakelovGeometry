noncomputable def fibreProbabilityMeasure (ω : AS.Ω) :
    Measure (ExtensionFiber AS L ω) :=
  Measure.sum fun x : ExtensionFiber AS L ω ↦ fibreWeight AS L ω x • Measure.dirac x
