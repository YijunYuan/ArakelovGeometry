noncomputable def spanPrime {q : Polynomial k} (hq : Prime q) :
    HeightOneSpectrum (Polynomial k) where
  asIdeal := Ideal.span {q}
  isPrime := (Ideal.span_singleton_prime hq.ne_zero).mpr hq
  ne_bot := by simpa [Ideal.span_singleton_eq_bot] using hq.ne_zero

@[simp]
