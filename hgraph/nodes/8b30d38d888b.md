theorem valuation_algebraMap_eq_one_of_ne {q : Polynomial k} (hq : Prime q)
    {v : HeightOneSpectrum (Polynomial k)} (hv : v ≠ spanPrime hq) :
    v.valuation (RatFunc k) (algebraMap (Polynomial k) (RatFunc k) q) = 1 := by
  rw [valuation_of_algebraMap, intValuation_eq_one_iff]
  intro hmem
  refine hv (HeightOneSpectrum.ext ?_)
  have hmax : (Ideal.span {q} : Ideal (Polynomial k)).IsMaximal :=
    Ideal.IsPrime.isMaximal ((Ideal.span_singleton_prime hq.ne_zero).mpr hq)
      (by simpa [Ideal.span_singleton_eq_bot] using hq.ne_zero)
  exact (hmax.eq_of_le v.isPrime.ne_top
    (Ideal.span_le.mpr (Set.singleton_subset_iff.mpr hmem))).symm.trans
    (spanPrime_asIdeal hq).symm
