private theorem measurable_base_eval (a : K) : Measurable (fun ω ↦ AS.ϕ ω a) := by
  by_cases ha : a = 0
  · simp [ha]
  · have h : (fun ω ↦ AS.ϕ ω a) = fun ω ↦ Real.exp (Real.log (AS.ϕ ω a)) := by
      funext ω
      exact (Real.exp_log ((AS.ϕ ω).pos ha)).symm
    rw [h]
    exact (AS.measurable_log a ha).exp
