theorem localHeight_pos {a : ι → K} (ha : a ≠ 0) (ω : AS.Ω) :
    0 < AS.localHeight a ω := by
  obtain ⟨i, hi⟩ : ∃ i, a i ≠ 0 := by
    by_contra! h
    exact ha (funext h)
  exact lt_of_lt_of_le ((AS.ϕ ω).pos hi) (AS.eval_le_localHeight a ω i)
