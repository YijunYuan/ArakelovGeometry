theorem measurable_κ {K : Type*} [Field K] (AS : AdelicStructure K) : Measurable AS.κ := by
  by_cases hC : CharZero K
  · -- char 0: rewrite κ by the bridge; `ω ↦ log (ϕ ω 2)` is measurable via `measurable_log`.
    have hκ : AS.κ = fun ω ↦ max 0 (Real.log ((AS.ϕ ω) 2)) / Real.log 2 :=
      funext (κ_eq_log_div AS)
    simpa [hκ] using
      (measurable_const.max (AS.measurable_log (2 : K) two_ne_zero)).div_const (Real.log 2)
  · -- not char 0: Ωinf = ∅, so κ is identically 0.
    simp [κ_eq_zero_of_not_charZero AS hC]
